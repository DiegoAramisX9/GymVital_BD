/* ---------- 4. PERMISOS (principio de MÍNIMO PRIVILEGIO) ---------- */

GRANT UNMASK TO rol_recepcion;                         -- ve email/teléfono reales para contactar al socio

DENY  DELETE ON SCHEMA::operacion TO rol_entrenador;

/* ---------- 6. AUDITORÍA DE CAMBIOS DDL (quién creó/alteró/borró objetos) ---------- */
CREATE OR ALTER TRIGGER trg_AuditarDDL ON DATABASE FOR DDL_DATABASE_LEVEL_EVENTS
AS
BEGIN
    SET NOCOUNT ON;
    DECLARE @e XML = EVENTDATA();
    INSERT auditoria.BitacoraDDL (Usuario, Evento, Objeto, Sentencia)
    VALUES (ORIGINAL_LOGIN(),
            @e.value('(/EVENT_INSTANCE/EventType)[1]',  'NVARCHAR(100)'),
            @e.value('(/EVENT_INSTANCE/SchemaName)[1]', 'NVARCHAR(100)') + N'.' + @e.value('(/EVENT_INSTANCE/ObjectName)[1]', 'NVARCHAR(200)'),
            @e.value('(/EVENT_INSTANCE/TSQLCommand/CommandText)[1]', 'NVARCHAR(MAX)'));
END
GO
GRANT SELECT ON SCHEMA::auditoria TO usr_admin_bd;   -- (db_owner ya lo tiene; se deja explícito para la matriz)
GO

/* ---------- 7. EVIDENCIA PARA EL INFORME: matriz de privilegios real ---------- */
-- 7.1 Pertenencia a roles
SELECT r.name AS Rol, m.name AS Usuario
FROM sys.database_role_members rm
JOIN sys.database_principals r ON r.principal_id = rm.role_principal_id
JOIN sys.database_principals m ON m.principal_id = rm.member_principal_id
WHERE r.name LIKE 'rol_%' OR r.name = 'db_owner'
ORDER BY r.name;

-- 7.2 Permisos efectivos declarados (GRANT/DENY) por rol
SELECT pr.name AS Rol, dp.state_desc AS Estado, dp.permission_name AS Permiso, dp.class_desc AS Clase,
       CASE dp.class WHEN 3 THEN SCHEMA_NAME(dp.major_id)
                     WHEN 1 THEN OBJECT_SCHEMA_NAME(dp.major_id) + '.' + OBJECT_NAME(dp.major_id) END AS Objeto,
       COL_NAME(dp.major_id, NULLIF(dp.minor_id, 0)) AS Columna
FROM sys.database_permissions dp
JOIN sys.database_principals pr ON pr.principal_id = dp.grantee_principal_id
WHERE pr.name LIKE 'rol_%'
ORDER BY pr.name, Objeto, dp.permission_name;
GO

/* =====================================================================
   Archivo : 04b_pruebas_seguridad.sql
   Responsable: Integrante 3 (Seguridad)
   Objetivo: DEMOSTRACIÓN DE ACCESO DENEGADO (diapositiva 4 de la defensa).
   Ejecutar DESPUÉS de 06_vistas_powerbi.sql.  Cada prueba cambia de contexto con
   EXECUTE AS USER, captura el error y vuelve con REVERT. Las pruebas permitidas
   se envuelven en una transacción con ROLLBACK para NO modificar datos.
   ===================================================================== */
USE GimnasioDB;
GO
SET NOCOUNT ON;

/* Procedimiento auxiliar de prueba: ejecuta una sentencia como otro usuario y reporta el resultado */
CREATE OR ALTER PROCEDURE dbo.usp_ProbarComo @Usuario SYSNAME, @Sentencia NVARCHAR(MAX), @Esperado VARCHAR(10)
AS
BEGIN
    DECLARE @sql NVARCHAR(MAX) = N'EXECUTE AS USER = ''' + @Usuario + N'''; BEGIN TRAN; ' + @Sentencia + N'; ROLLBACK; REVERT;';
    DECLARE @res VARCHAR(10) = 'PERMITIDO', @msg NVARCHAR(500) = N'';
    BEGIN TRY
        EXEC (@sql);
    END TRY
    BEGIN CATCH
        SET @res = 'DENEGADO'; SET @msg = ERROR_MESSAGE();
        IF XACT_STATE() <> 0 ROLLBACK;
        REVERT;
    END CATCH
    SELECT @Usuario AS Usuario, LEFT(@Sentencia, 70) AS Sentencia, @Esperado AS Esperado, @res AS Obtenido,
           IIF(@Esperado = @res, 'OK', 'FALLO') AS Veredicto, @msg AS MensajeDeError;
END
GO

PREGUNTAS PROBABLES DE DEFENSA
==============================================================================
  - Por que roles y no permisos directos? -> Mantenimiento: auditar y dar de baja accesos es mas facil y menos propenso a error.
  - Que es el minimo privilegio? -> Dar solo los permisos necesarios para la tarea.
  - Diferencia entre DENY y REVOKE? -> DENY bloquea aunque haya GRANT por otro rol; REVOKE solo retira un GRANT/DENY anterior.
  - Como puede Power BI leer sin permisos sobre las tablas? -> Encadenamiento de propiedad: vista y tablas tienen el mismo dueno (dbo).
  - Que hace el enmascaramiento dinamico? -> Muestra datos ofuscados a quien no tiene UNMASK; no cambia el dato guardado.
